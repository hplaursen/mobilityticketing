-- Demonstrate that an old query no longer compiles after removing its column.
-- The whole change is rolled back after the expected error is caught.
begin;
BEGIN
alter table tickets drop column product_code;
ALTER TABLE
do $$
begin
    begin
        execute 'select product_code from tickets limit 1';
        raise exception 'Expected the legacy query to fail';
    exception when undefined_column then
        raise notice 'Expected failure: old reader refers to removed product_code';
    end;
end;
$$;
DO
rollback;

===== database/postgres/migrations/030_expand_product_identity.sql =====
-- Expand: add stable product UUIDs while keeping product_code usable.
begin;
BEGIN
set local lock_timeout = '3s';
SET
alter table products
    add column id uuid default gen_random_uuid();
ALTER TABLE
alter table products
    alter column id set not null,
    add constraint products_id_unique unique (id);
ALTER TABLE
alter table tickets
    add column product_id uuid;
ALTER TABLE
alter table tickets
    add constraint tickets_product_id_fk
    foreign key (product_id) references products(id) not valid;
ALTER TABLE
-- Prevent accidental reassignment after the UUID has been issued.
create function prevent_product_id_change()
returns trigger language plpgsql as $$
begin
    if new.id is distinct from old.id then
        raise exception 'Product ID is immutable';
    end if;
    return new;
end;
$$;
CREATE FUNCTION
create trigger products_id_immutable
before update of id on products
for each row execute function prevent_product_id_change();
CREATE TRIGGER
commit;
COMMIT

begin;
BEGIN
insert into tickets
    (id, user_id, trip_id, ticket_code, status, product_code, product_id,
     valid_from_utc, valid_to_utc, price, currency)

       'Active', single_product.code, day_product.id,
       t.valid_from_utc, t.valid_to_utc, t.price, t.currency
from tickets t
cross join products single_product
cross join products day_product
where t.id = 'TICKET-1'
  and single_product.code = 'SINGLE'
  and day_product.code = 'DAY';
INSERT 0 1
do $$
begin
    if exists (select 1 from tickets where id = 'LAB04-MISMATCH') then
        raise notice 'Database accepted the mismatched pair; writers and verification must prevent it';
    end if;
end;

===== database/postgres/experiments/lecture04/new_writer.sql =====
-- New writer accepts a product ID, derives its code, and keeps the agreed
-- purchase price as an independent input (not the current catalogue price).
\set ticket_id 'LAB04-NEW-1'
\set ticket_code 'LAB04-CODE-NEW-1'
select id as product_id from products where code = 'DAY' \gset
insert into tickets
    (id, user_id, trip_id, ticket_code, status, product_code, product_id,
     valid_from_utc, valid_to_utc, price, currency)
select :'ticket_id', source.user_id, source.trip_id, :'ticket_code',
       'Active', p.code, p.id,
       source.valid_from_utc, source.valid_to_utc, 65.00, 'DKK'
from tickets source
join products p on p.id = :'product_id'::uuid
where source.id = 'TICKET-1';
INSERT 0 1
-- A caller supplies SINGLE with the DAY product ID. The writer ignores the
-- conflicting supplied code and stores the code looked up from the ID.
with request(supplied_code, product_id) as (
    values ('SINGLE'::text, :'product_id'::uuid)
)
insert into tickets
    (id, user_id, trip_id, ticket_code, status, product_code, product_id,
     valid_from_utc, valid_to_utc, price, currency)
select 'LAB04-NEW-CONFLICT', source.user_id, source.trip_id,
       'LAB04-CODE-NEW-CONFLICT', 'Active',
       case when request.supplied_code is distinct from p.code
            then p.code else request.supplied_code end,
       p.id, source.valid_from_utc, source.valid_to_utc, 65.00, 'DKK'
from request
join products p on p.id = request.product_id
join tickets source on source.id = 'TICKET-1';
INSERT 0 1
-- Unknown IDs are rejected explicitly.
do $$
declare inserted_rows integer;
begin
    insert into tickets
        (id, user_id, trip_id, ticket_code, status, product_code, product_id,
         valid_from_utc, valid_to_utc, price, currency)
    select 'LAB04-UNKNOWN-PRODUCT', source.user_id, source.trip_id,
           'LAB04-CODE-UNKNOWN-PRODUCT', 'Active', p.code, p.id,
           source.valid_from_utc, source.valid_to_utc, 65.00, 'DKK'
    from tickets source
    join products p on p.id = '00000000-0000-0000-0000-000000000000'::uuid
    where source.id = 'TICKET-1';
    get diagnostics inserted_rows = row_count;
    if inserted_rows = 0 then
        raise notice 'Unknown product ID rejected';
    else
        raise exception 'Unknown product ID unexpectedly inserted a ticket';
    end if;
end;
$$;
DO
